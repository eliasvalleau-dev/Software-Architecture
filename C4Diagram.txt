workspace "Course Management System" "Architectural Design and C4 Models" {

    !identifiers hierarchical

    model {
        # Actors
        student = person "Student" "Enrolls in subjects and tracks academic standing."
        faculty = person "Faculty Member" "Manages subject offerings and reviews academic analytics."
        admin = person "System Administrator" "Manages system access permissions and monitors operational health."

        # External Software Systems
        idp = softwareSystem "Okta Identity & Access Management" "External Okta service that authenticates users and supplies signed role or group claims." {
            tags "External IAM"
        }
        sis = softwareSystem "Student Information System" "Existing system holding master profiles, academic standing, subject catalogue, and official academic grades." {
            tags "External SIS"
        }
        analytics = softwareSystem "Amazon QuickSight" "AWS-managed business intelligence service for visualising aggregated academic performance and enrolment trends." {
            tags "External Reporting"
        }

        # Primary System Boundary
        cms = softwareSystem "Course Management System" "Provides real-time subject-offering enrolment, academic-grade tracking, and reporting analytics tools." {
            tags "Primary System"

            # Level 2 Containers
            spa = container "Single-Page Application" "Provides subject-offering enrolment, availability tracking, and academic reporting interfaces through a web browser." "React / TypeScript" {
                tags "UI Container"
            }

            backend = container "Backend" "Processes enrolments and subject offerings, serves user-requested academic summaries, and generates PDF reports." "Java / Spring Boot" {
                tags "Backend Container"

                # Level 3 Application Components
                enrolmentApi = component "Enrolment API" "Accepts enrolment requests, checks offering capacity, manages waitlisting, and queues requests during peak demand." "Spring MVC Controller" {
                    tags "Component"
                }

                subjectOfferingApi = component "Subject Offering API" "Manages subject offerings, including term, section, capacity, coordinator, enrolment window, and availability." "Spring MVC Controller" {
                    tags "Component"
                }

                analyticsApi = component "Analytics API" "Receives user requests for academic performance summaries and downloadable reports." "Spring MVC Controller" {
                    tags "Component"
                }

                securityComponent = component "Security Component" "Intercepts requests, validates signed Okta access tokens and claims, and enforces session authorisation." "Spring Security" {
                    tags "Component"
                }

                reportingComponent = component "Reporting Component" "Generates user-requested academic summaries and PDF reports from operational data." "Spring Bean" {
                    tags "Component"
                }

                sisAdapter = component "SIS Adapter" "Encapsulates REST client integration with the external Student Information System." "Spring Bean" {
                    tags "Component"
                }

                # P2SDS Logical Data Model
                userEntity = component "User" "Stores shared identity and contact details, including the Okta identity-provider subject reference." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                studentEntity = component "Student" "Stores student ID, degree program, and academic standing. Uses User ID as its primary and foreign key." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                facultyEntity = component "Faculty" "Stores faculty ID and faculty type. Uses User ID as its primary and foreign key." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                subjectEntity = component "Subject" "Stores catalogue subject code, name, credit points, and an optional simple prerequisite subject." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                subjectOfferingEntity = component "Subject Offering" "Stores one term-specific subject instance, including section, coordinator, capacity, enrolment window, and availability." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                enrolmentEntity = component "Enrolment" "Links a student to a subject offering and records Enrolled, Waitlisted, or Dropped status, timestamps, and waitlist sequence." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }

                academicGradeEntity = component "Academic Grade" "Stores the final numerical and letter grade for an enrolment." "JPA Entity / Logical Data Model" {
                    tags "Component"
                }
            }

            # Separate scheduled analytics pipeline; not part of interactive enrolment or reporting.
            etl = container "Data Ingestion & ETL" "Extracts approved operational data, validates and aggregates it, and exports datasets for external analytics." "Java / Spring Batch (scheduled ECS Fargate task)" {
                tags "Backend Container"

                extractor = component "Data Extractor" "Retrieves approved academic and enrolment records using read-only queries." "Spring Batch Reader" {
                    tags "Component"
                }

                transformer = component "Data Transformer" "Validates, cleans, de-identifies where appropriate, and aggregates extracted data." "Spring Batch Processor" {
                    tags "Component"
                }

                exportHandler = component "Export Handler" "Publishes validated analytics extracts to S3 and records export status." "Spring Batch Writer" {
                    tags "Component"
                }
            }

            cache = container "In-Memory Cache & Queue" "Caches active subject-offering availability and buffers incoming enrolment requests." "Redis" {
                tags "Cache Store"
            }

            db = container "Operational Relational Database" "Stores Users, Students, Faculty, Subjects, Subject Offerings, Enrolments, and Academic Grades." "Amazon RDS PostgreSQL (Multi-AZ)" {
                tags "Database Store"
            }

            storage = container "Report & Data Export Storage" "Stores generated PDF reports, approved analytics extracts, and exported academic backups in separate access-controlled locations." "Amazon S3 Bucket" {
                tags "S3 Bucket Store"
            }

            # Level 2 Container Relationships
            spa -> backend "Sends API requests to" "JSON / HTTPS"
            backend -> cache "Reads subject-offering availability and queues enrolment requests in" "Redis Protocol"
            backend -> db "Reads and writes P2SDS records to" "SQL / JDBC"
            backend -> storage "Saves and retrieves user-requested PDF reports" "AWS S3 API / HTTPS"
            etl -> db "Extracts approved academic records using read-only queries" "SQL / JDBC (read-only)"
            etl -> storage "Publishes transformed analytics datasets" "AWS S3 API / HTTPS"

            # Level 3 Component Relationships
            spa -> backend.enrolmentApi "Submits enrolment requests and checks seat counts via" "JSON / HTTPS"
            spa -> backend.subjectOfferingApi "Queries subject details and offered sections via" "JSON / HTTPS"
            spa -> backend.analyticsApi "Requests academic performance reports via" "JSON / HTTPS"

            backend.enrolmentApi -> backend.securityComponent "Delegates access-token validation to"
            backend.subjectOfferingApi -> backend.securityComponent "Delegates access-token validation to"
            backend.analyticsApi -> backend.securityComponent "Delegates access-token validation to"

            # Enrolment and subject-offering behaviour
            backend.enrolmentApi -> backend.enrolmentEntity "Creates and updates"
            backend.enrolmentApi -> backend.studentEntity "Checks student standing for"
            backend.enrolmentApi -> backend.subjectOfferingEntity "Checks capacity and availability for"
            backend.enrolmentApi -> backend.subjectEntity "Checks simple prerequisite for"
            backend.enrolmentApi -> cache "Reads seat availability and buffers write requests in" "Redis Protocol"
            backend.enrolmentApi -> db "Persists enrolments and waitlist state to" "SQL / JDBC"
            backend.enrolmentApi -> backend.sisAdapter "Verifies transcript eligibility and synchronises enrolments via"

            backend.subjectOfferingApi -> backend.subjectOfferingEntity "Manages"
            backend.subjectOfferingApi -> backend.subjectEntity "Reads catalogue and prerequisite data from"
            backend.subjectOfferingApi -> cache "Caches updated subject offerings in" "Redis Protocol"
            backend.subjectOfferingApi -> db "Reads and writes subject and offering records in" "SQL / JDBC"

            # Analytics and reporting
            backend.analyticsApi -> backend.reportingComponent "Delegates report compilation to"
            backend.reportingComponent -> backend.academicGradeEntity "Queries performance data from"
            backend.reportingComponent -> db "Queries aggregate student performance data from" "SQL / JDBC"
            backend.reportingComponent -> storage "Saves generated PDF report files to" "AWS S3 API / HTTPS"

            # ETL components: independent scheduled data-processing workflow
            etl.extractor -> db "Reads approved source records from" "SQL / JDBC (read-only)"
            etl.extractor -> etl.transformer "Passes extracted records to"
            etl.transformer -> etl.exportHandler "Passes validated and aggregated datasets to"
            etl.exportHandler -> storage "Writes analytics extracts to" "AWS S3 API / HTTPS"

            # P2SDS logical entity relationships
            backend.studentEntity -> backend.userEntity "Uses User ID as PK/FK"
            backend.facultyEntity -> backend.userEntity "Uses User ID as PK/FK"
            backend.subjectOfferingEntity -> backend.subjectEntity "Is one scheduled offering of"
            backend.subjectOfferingEntity -> backend.facultyEntity "Has one coordinator"
            backend.enrolmentEntity -> backend.studentEntity "References"
            backend.enrolmentEntity -> backend.subjectOfferingEntity "References"
            backend.academicGradeEntity -> backend.enrolmentEntity "Is produced from"
            backend.subjectEntity -> backend.subjectEntity "May require one prerequisite subject"

            # Security and external integration
            backend.securityComponent -> cache "Validates cached session state in" "Redis Protocol"
            backend.securityComponent -> idp "Uses Okta for user authentication and signed role or group claims via" "OAuth2 / OIDC"
            backend.sisAdapter -> sis "Calls REST endpoints on" "HTTPS / REST API"

            
            cms.storage -> analytics "Provides transformed analytics datasets for dashboards" "Amazon S3"


            #Dynamic Diagram Information
            spa -> backend.enrolmentApi "Submits enrolment request to" "JSON/HTTPS"
            backend.enrolmentApi -> spa "Validates access token and permissions " "JSON/HTTPS"
            backend.enrolmentApi -> backend.securityComponent "Validates request using" 
            backend.securityComponent -> backend.enrolmentApi "Issues an access token if request succeeds" 
            backend.securityComponent -> db "select * from users where = ?" "SQL Query"
            db -> backend.securityComponent "Returns user enrolment data to" "SQL Query"
        }

        # External Relationships: Actors to Primary System Containers
        student -> cms.spa "Selects subject offerings and checks availability via" "HTTPS"
        faculty -> cms.spa "Manages subject offerings and views analytics via" "HTTPS"
        admin -> cms.spa "Manages system configurations via" "HTTPS"

        # Production deployment model. AWS delivery and operations services are
        # infrastructure nodes; the C4 containers above remain the logical model.
        deploymentEnvironment "Production" {
            clientDevices = deploymentNode "Student, Faculty and Administrator Devices" "Web browsers running the Course Management System SPA." "Client" {
                spaInstance = containerInstance cms.spa
            }

            externalServices = deploymentNode "External Identity, Student Information and Analytics Services" "Third-party university and SaaS systems." "External Services" {
                idpInstance = softwareSystemInstance idp
                sisInstance = softwareSystemInstance sis
                analyticsInstance = softwareSystemInstance analytics
            }

            awsRegion = deploymentNode "AWS Region - Sydney" "ap-southeast-2 production deployment." "AWS" {
                edgeServices = deploymentNode "AWS Edge Services" "Public application delivery." "AWS Managed" {
                    waf = infrastructureNode "AWS WAF" "Web ACL associated with CloudFront to inspect and block malicious HTTP traffic." "AWS WAF"
                    cloudfront = infrastructureNode "CloudFront Distribution" "Public entry point for static assets and API requests." "Amazon CloudFront"
                }

                regionalServices = deploymentNode "Regional Storage and Operations Services" "Managed AWS services outside the CMS VPC." "AWS Managed" {
                    frontendAssets = infrastructureNode "Frontend S3 Bucket" "Private bucket for compiled React assets, protected by Origin Access Control." "Amazon S3"
                    reportStorage = deploymentNode "Report and Data Export S3" "Private storage for generated PDF reports and approved analytics extracts." "Amazon S3" {
                        storageInstance = containerInstance cms.storage
                    }
                    secrets = infrastructureNode "AWS Secrets Manager" "Stores application credentials and Okta client secrets." "AWS Secrets Manager"
                    logging = infrastructureNode "Amazon CloudWatch Logs" "Collects application logs and operational events." "Amazon CloudWatch"
                    registry = infrastructureNode "Amazon ECR" "Stores backend container images for deployment." "Amazon ECR"
                    backups = infrastructureNode "RDS Automated Backups" "Point-in-time recovery and agreed backup-retention policy for the operational database." "Amazon RDS"
                }

                cmsVpc = deploymentNode "CMS VPC" "10.0.0.0/16." "Amazon VPC" {
                    tags "VPC"

                    alb = infrastructureNode "Internet-facing Application Load Balancer" "Accepts HTTPS requests from CloudFront and distributes them to healthy backend tasks." "AWS ALB"

                    availabilityZoneA = deploymentNode "Availability Zone A" "First availability zone." "AWS AZ" {
                        tags "AZ"

                        publicSubnetA = deploymentNode "Public Subnet A" "10.0.0.0/24 - zonal egress." "Subnet" {
                            tags "Public Subnet"
                            natGatewayA = infrastructureNode "NAT Gateway A" "Zonal HTTPS egress for private application subnet A." "AWS NAT Gateway"
                        }

                        privateApplicationSubnetA = deploymentNode "Private Application Subnet A" "10.0.20.0/24." "Subnet" {
                            tags "Private Subnet"
                            backendA = containerInstance cms.backend
                        }

                        privateDataSubnetA = deploymentNode "Private Data Subnet A" "10.0.30.0/24." "Subnet" {
                            tags "Private Subnet"
                            rdsPrimary = containerInstance cms.db
                            redisPrimary = containerInstance cms.cache
                        }
                    }

                    availabilityZoneB = deploymentNode "Availability Zone B" "Second availability zone." "AWS AZ" {
                        tags "AZ"

                        publicSubnetB = deploymentNode "Public Subnet B" "10.0.1.0/24 - zonal egress." "Subnet" {
                            tags "Public Subnet"
                            natGatewayB = infrastructureNode "NAT Gateway B" "Zonal HTTPS egress for private application subnet B." "AWS NAT Gateway"
                        }

                        privateApplicationSubnetB = deploymentNode "Private Application Subnet B" "10.0.21.0/24." "Subnet" {
                            tags "Private Subnet"
                            backendB = containerInstance cms.backend
                        }

                        privateDataSubnetB = deploymentNode "Private Data Subnet B" "10.0.31.0/24." "Subnet" {
                            tags "Private Subnet"
                            rdsStandby = infrastructureNode "RDS Multi-AZ Standby" "Synchronous standby with automatic failover; no direct application traffic." "Amazon RDS PostgreSQL" {
                                tags "Standby"
                            }
                            redisReplica = infrastructureNode "Redis Replica" "Replication target with automatic failover; no direct application traffic." "Amazon ElastiCache Redis" {
                                tags "Standby"
                            }
                        }
                    }

                    vpcEndpoints = infrastructureNode "VPC Endpoints" "Gateway endpoint for S3 and interface endpoints for ECR, Secrets Manager, and CloudWatch Logs." "AWS PrivateLink / Gateway"
                }
            }

            clientDevices.spaInstance -> awsRegion.edgeServices.cloudfront "Loads application assets and invokes API endpoints through" "HTTPS 443"
            awsRegion.edgeServices.waf -> awsRegion.edgeServices.cloudfront "Protects"
            awsRegion.edgeServices.cloudfront -> awsRegion.regionalServices.frontendAssets "Retrieves React assets from" "HTTPS 443"
            awsRegion.edgeServices.cloudfront -> awsRegion.cmsVpc.alb "Forwards API requests to" "HTTPS 443"
            awsRegion.cmsVpc.alb -> awsRegion.cmsVpc.availabilityZoneA.privateApplicationSubnetA.backendA "Routes requests to healthy tasks" "HTTPS 8443"
            awsRegion.cmsVpc.alb -> awsRegion.cmsVpc.availabilityZoneB.privateApplicationSubnetB.backendB "Routes requests to healthy tasks" "HTTPS 8443"
            awsRegion.regionalServices.registry -> awsRegion.cmsVpc.availabilityZoneA.privateApplicationSubnetA.backendA "Supplies container image during deployment" "HTTPS 443"
            awsRegion.regionalServices.registry -> awsRegion.cmsVpc.availabilityZoneB.privateApplicationSubnetB.backendB "Supplies container image during deployment" "HTTPS 443"
            awsRegion.cmsVpc.availabilityZoneA.privateApplicationSubnetA.backendA -> awsRegion.regionalServices.secrets "Retrieves application secrets" "HTTPS 443"
            awsRegion.cmsVpc.availabilityZoneB.privateApplicationSubnetB.backendB -> awsRegion.regionalServices.secrets "Retrieves application secrets" "HTTPS 443"
            awsRegion.cmsVpc.availabilityZoneA.privateApplicationSubnetA.backendA -> awsRegion.regionalServices.logging "Sends application logs" "HTTPS 443"
            awsRegion.cmsVpc.availabilityZoneB.privateApplicationSubnetB.backendB -> awsRegion.regionalServices.logging "Sends application logs" "HTTPS 443"
        }
    }
